   showEpitopeLabels();
   pdbViewer.zoomTo(epi,700);pdbViewer.render();
   setPdbInfo('<strong>Resíduos do epítopo.</strong><br>Esta visualização mostra especificamente os resíduos da lisozima nos segmentos <span class="mono">18–27</span> e <span class="mono">117–125</span>, cada um com seu nome e número.  }else if(mode==="interfaceatoms"){
   basePdbStyle();
   pdbViewer.setStyle(iAg,{sphere:{color:"#fb7185",scale:.30},stick:{color:"#fb7185",radius:.10}});

 }
}
function initPDBViewer(){
 if(pdbReady){try{pdbViewer.resize();pdbViewer.render()}catch(e){}return}
 const loading=document.getElementById("pdbLoading");
 if(typeof $3Dmol==="undefined"){
   loading.innerHTML='<div><strong>Visualizador 3D indisponível.</strong><br><span style="color:#7f93ad">Esta etapa requer internet para carregar a biblioteca 3Dmol.js.</span></div>';return;
 }
 try{
   pdbViewer=$3Dmol.createViewer("pdbViewer",{backgroundColor:"#06101c",antialias:true});
   $3Dmol.download("pdb:1FDL",pdbViewer,{doAssembly:false},function(){
     const detected=detectPdbChains();
     if(!detected.atoms){
       loading.innerHTML='<div><strong>O arquivo PDB foi recebido, mas nenhum átomo foi reconhecido.</strong></div>';return;
     }
     updateEpitopeResidueList();
     pdbReady=true;basePdbStyle();pdbViewer.resize();pdbViewer.zoomTo({});pdbViewer.render();loading.classList.add("hide");
     setPdbInfo('<strong>Estrutura carregada.</strong><br>PDB <span class="mono">1FDL</span>: '+detected.atoms+' átomos reconhecidos. Cadeias do arquivo: <span class="mono">'+detected.chains.join(", ")+'</span>.<br><br>Gire o complexo e depois selecione <strong>Destacar epítopo</strong>.');
   });
   setTimeout(()=>{if(!pdbReady)loading.innerHTML='<div><strong>A estrutura está demorando para carregar.</strong><br><span style="color:#7f93ad">Verifique a conexão. O restante da atividade continua funcionando.</span></div>'},12000);
 }catch(err){loading.innerHTML='<div><strong>Falha ao iniciar o visualizador.</strong><br><span style="color:#7f93ad">'+String(err.message||err)+'</span></div>'}
}
