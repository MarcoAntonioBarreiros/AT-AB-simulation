function clearPdbLabels(){
  if(!pdbViewer)return;
  try{pdbViewer.removeAllLabels()}catch(e){}
}
function epitopeResiduesFromModel(){
  if(!pdbViewer)return [];
  const atoms=pdbViewer.selectedAtoms(epitopeSel());
  const map=new Map();
  atoms.forEach(a=>{
    const key=(a.chain||"")+"|"+a.resi;
    if(!map.has(key)) map.set(key,{chain:a.chain,resi:+a.resi,resn:a.resn||"RES",atom:a.atom==="CA"?a:null,first:a});
    else if(a.atom==="CA") map.get(key).atom=a;
  });
  return [...map.values()].sort((a,b)=>a.resi-b.resi);
}
function showEpitopeLabels(){

  epitopeResiduesFromModel().forEach(r=>{

    pdbViewer.addLabel((r.resn||"RES")+r.resi,{
      position:{x:a.x,y:a.y,z:a.z},

      fontColor:"#ffd7df",
      fontSize:10,

    });
  });
}

function detectPdbChains(){
  const atoms=pdbViewer.selectedAtoms({});
  const chains=[...new Set(atoms.map(a=>a.chain).filter(Boolean))];
  if(chains.includes("Y")&&chains.includes("L")&&chains.includes("H")){
    pdbChains={antigen:"Y",light:"L",heavy:"H"};
  }else if(chains.includes("C")&&chains.includes("A")&&chains.includes("B")){
    pdbChains={antigen:"C",light:"A",heavy:"B"};
  }else if(chains.length>=3){
    // Fallback only to avoid an empty viewer; 1FDL is expected to match one of the maps above.
    pdbChains={light:chains[0],heavy:chains[1],antigen:chains[2]};
  }
  return {atoms:atoms.length,chains};
}

}
